```gdscript
func get_objectives() -> Array[Dictionary]:

    for b in get_tree().get_nodes_in_group("enemy_buildings"):

    return false
```

```gdscript
func _add_stats_section(parent: Container, is_victory: bool) -> void:
    var stats_data := _collect_game_stats(is_victory)
    for item in stats_data:
        var row := HBoxContainer.new()
        row.alignment = BoxContainer.ALIGNMENT_CENTER
        var label := Label.new()
        label.text = item.label + ": "
        label.add_theme_font_size_override("font_size", 14)
        row.add_child(label)
        var value := Label.new()
        value.text = str(item.value)
        value.add_theme_font_size_override("font_size", 14)
        value.add_theme_color_override("font_color", Color(0.4, 1.0, 0.4))
        row.add_child(value)
        parent.add_child(row)


# T3 PR-3: 收集本局统计数据
func _collect_game_stats(is_victory: bool) -> Array:
    var time_used := _game_time - _game_start_time
    var minutes := int(time_used) / 60
    var seconds := int(time_used) % 60
    var time_str := "%d:%02d" % [minutes, seconds]
    return [
        {"label": "用时", "value": time_str},
        {"label": "剩余金币", "value": gold},
        {"label": "获得金币总数", "value": _gold_earned},
        {"label": "击杀敌方单位", "value": _enemies_killed},
        {"label": "损失单位", "value": _units_lost},
        {"label": "建造单位数", "value": _units_trained},
        {"label": "建造建筑数", "value": _buildings_constructed},
    ]
```

## 三、数据流图

### 3.1 战斗统计追踪

```
战斗事件                埋点位置                       统计字段
─────────────────────────────────────────────────────────
玩家击杀敌方单位  ←  _on_unit_died (enemy_units)  →  _enemies_killed
玩家损失单位      ←  _on_unit_died (player_units) →  _units_lost (现有)
玩家造单位        ←  game_spawner                 →  _units_trained
玩家造建筑        ←  building_placer              →  _buildings_constructed
玩家获得金币      ←  add_gold(amount > 0)         →  _gold_earned
游戏用时          ←  _game_time - _game_start_time
剩余金币          ←  gold
```

### 3.2 胜利流程

```
玩家摧毁敌方城堡（victory_destroy_base.check() 返回 1）
  ↓
VictoryCondition.game_ended.emit("victory")
  ↓
main._on_game_ended("victory")
  ↓
_show_victory_panel(sm)
  ↓
_add_stats_section(vbox, true)
  ↓
_collect_game_stats(true) → 显示 7 项
  ↓
玩家点"返回菜单"
```

### 3.3 objectives_panel 任务显示

```
victory_destroy_base.get_objectives()
  ↓ 返回 [{text: "摧毁敌方据点", state: 0/1/2}, ...]
objectives_panel._update_objectives() (每秒调用)
  ↓
_obj_container 渲染 Label
  ↓
state=0 灰白色 / state=1 绿色 / state=2 红色
```

### 3.4 开局提示流程

```
main._run_init_steps() 全部完成
  ↓
实例化 IntroHintDialog
  ↓
玩家看到居中提示"本局任务: 摧毁敌方据点..."
  ↓
玩家点"知道了"
  ↓
queue_free()
```

## 六、测试要点

### 6.1 战斗统计准确性

1. 进游戏，记录初始统计（应该全 0，_gold_earned 取决于初始金币是否走 add_gold）
2. 造一个农场（-100 金）→ 验证 `_buildings_constructed == 1`，`_gold_earned` 不变（支出不算）
3. 农场完成反还 100 金 → 验证 `_gold_earned == 100`
4. 用兵营造 2 个步兵 → 验证 `_units_trained == 2`
5. 让步兵去杀 1 个敌方单位 → 验证 `_enemies_killed == 1`
6. 让 1 个步兵被杀 → 验证 `_units_lost == 1`，`_enemies_killed` 不变

### 6.2 objectives_panel 显示

1. 进游戏后，右上角 Info 面板 Objectives 区显示：
   - "摧毁敌方据点（敌方城堡）"（灰白色，进行中）
   - "保护我方城堡"（灰白色，进行中）
2. 玩家城堡被打到一半 HP → 任务状态不变（state 还是 0）
3. 玩家城堡被毁 → "保护我方城堡"变红色（state=2）
4. 敌方城堡被毁 → "摧毁敌方据点"变绿色（state=1）

### 6.3 胜利统计画面

1. 摧毁敌方城堡 → 胜利画面弹出
2. 验证显示 7 项：用时 / 剩余金币 / 获得金币总数 / 击杀敌方单位 / 损失单位 / 建造单位数 / 建造建筑数
3. 数据准确
4. 点"返回菜单"能正常跳转

### 6.4 失败统计画面

1. 让玩家城堡被毁 → 失败画面弹出（红色标题"失败"）
2. 同样显示 7 项统计
3. 数据准确

### 6.5 开局一次性提示

1. 进游戏后看到居中提示"本局任务: 摧毁敌方据点..."
2. 按 ESC 无效（强制看）
3. 点"知道了" → 提示消失
4. 关卡内不再出现（除非重进游戏）

1. 升 T2 / T3，造学院，研究 1 个 T3 升级（PR-2 内容）
2. 用变体单位攻打敌方城堡
